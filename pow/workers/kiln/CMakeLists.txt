cmake_minimum_required(VERSION 3.16)
project(igneum-kiln-vk CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# the reference library (pow/ref) is built as a subdirectory so the worker links the same consensus truth it is measured against
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../ref ${CMAKE_CURRENT_BINARY_DIR}/ref)
find_package(Vulkan REQUIRED)
find_package(Threads REQUIRED)
find_library(SHADERC_LIB NAMES shaderc shaderc_shared REQUIRED)
find_path(SHADERC_INC shaderc/shaderc.h REQUIRED)
add_executable(kiln_vk kiln_vk.cpp)
target_include_directories(kiln_vk PRIVATE ${SHADERC_INC} ${CMAKE_CURRENT_SOURCE_DIR}/../../ref/include ${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/blake3-1.8.2)
target_link_libraries(kiln_vk PRIVATE ignpow blake3 Vulkan::Vulkan ${SHADERC_LIB} Threads::Threads)
target_compile_options(kiln_vk PRIVATE -O2 -Wall -Wextra)
